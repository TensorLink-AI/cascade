ALLOWED = ("from_block", "share", "takes_from", "remainder")


def parse_lane_d(raw: dict) -> dict:
    out = {"from_block": int(raw.get("from_block", 0))}
    out["share"] = float(raw["share"])
    if "takes_from" in raw:
        out["takes_from"] = str(raw["takes_from"])
    out["remainder"] = str(raw.pop("remainder"))
    return out


def parse_lane_d_ops(raw: dict) -> dict:
    return {"poll_seconds": int(raw.get("poll_seconds", 120))}
